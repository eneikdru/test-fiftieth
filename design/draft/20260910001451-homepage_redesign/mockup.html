export async function persistAuthenticatedSession(claims: JWTClaims) {
  // Atomic upsert linking verified token claims to PostgreSQL
  return await db.transaction(async (tx) => {
    const [user] = await tx.insert(users).values({
      id: claims.sub,
      email: claims.email,
      roles: claims['https://purl.imsglobal.org/spec/lti/claim/roles']
    }).onConflictDoUpdate({ target: users.id, set: { lastSeen: new Date() } }).returning();

    await tx.insert(sessions).values({
      userId: user.id,
      tokenHash: sha256(claims.jti),
      expiresAt: new Date(claims.exp * 1000)
    });
  });
}
